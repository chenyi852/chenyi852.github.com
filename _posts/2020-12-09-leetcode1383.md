## 思路

公司有编号为 1 到 n 的 n 个工程师，给你两个数组 speed 和 efficiency ，其中 speed[i] 和 efficiency[i] 分别代表第 i 位工程师的速度和效率。请你返回由最多 k 个工程师组成的 最大团队表现值 ，由于答案可能很大，请你返回结果对 10^9 + 7 取余后的结果。

团队表现值 的定义为：一个团队中「所有工程师速度的和」乘以他们「效率值中的最小值」。

来源：力扣（LeetCode）
链接：https://leetcode-cn.com/problems/maximum-performance-of-a-team
著作权归领扣网络所有。商业转载请联系官方授权，非商业转载请注明出处。

示例 1：

```
输入：n = 6, speed = [2,10,3,1,5,8], efficiency = [5,4,3,9,7,2], k = 2
输出：60
解释：
我们选择工程师 2（speed=10 且 efficiency=4）和工程师 5（speed=5 且 efficiency=7）。他们的团队表现值为 performance = (10 + 5) * min(4, 7) = 60 。
```

示例 2：
```
输入：n = 6, speed = [2,10,3,1,5,8], efficiency = [5,4,3,9,7,2], k = 3
输出：68
解释：
此示例与第一个示例相同，除了 k = 3 。我们可以选择工程师 1 ，工程师 2 和工程师 5 得到最大的团队表现值。表现值为 performance = (2 + 10 + 5) * min(5, 4, 7) = 68 。
```
示例 3：
```
输入：n = 6, speed = [2,10,3,1,5,8], efficiency = [5,4,3,9,7,2], k = 4
输出：72
```

先按照efficiency的顺序排序，大的排在前面，我们尽可能让efficency高的工程师排在前面。

按照题目给的公司计算performance,并于历史的performance比较，求出最大的performance值。 当计算的工程师超过K个时，则从优先级队列中弹出speed最小的工程师，这样可以保证speed的和永远最大，并将speed从speed sum中减去。





```c
typedef struct _engineer_t {
    int speed;
    int efficiency;
} engineer_t;

static engineer_t *engineer_create(int n, int *speed, int *efficiency) {
    engineer_t *en = (engineer_t *)malloc(sizeof(engineer_t) * n);
    for (int i = 0; i < n; i++) {
        en[i].speed = speed[i];
        en[i].efficiency = efficiency[i];
    }
    return en;
}

static void engineer_destory(engineer_t *en) {
    free(en);
}

static int com(const void *a, const void *b) {
    engineer_t *en_a = (engineer_t *)a;
    engineer_t *en_b = (engineer_t *)b;

    return (en_b->efficiency - en_a->efficiency);
}

#define QUEUE_SZ    100000
typedef struct _pri_queue_t {
    int top;
    int data[QUEUE_SZ];
} pri_queue_t;

static pri_queue_t q;
static void queue_init(void) {
    q.top = 0;
    q.data[0] = INT_MIN;
}

static void queue_push(int d) {
    int i;
    for (i = ++q.top; d < q.data[i / 2]; i /= 2) {
        q.data[i]  = q.data[i / 2]; 
    } 

    q.data[i] =  d; 
}

static int queue_pop(void) {
    int ret = q.data[1];
    int last_element = q.data[q.top--];
    int child = 0;
    int i;

    for (i = 1; i * 2 <= q.top; i = child) {
        child = i * 2;
        if (child != q.top && q.data[child] > q.data[child + 1]) {
            child++;
        }

        if (q.data[child] < last_element) {
            q.data[i] = q.data[child];
        } else {
            break;
        }
    }
    q.data[i] = last_element;
    return ret;
}

static int queue_size(void) {
    return (q.top);
}

static long long max(long long a, long long b) {
    return (a > b ? a :b);
}

int maxPerformance(int n, int* speed, int speedSize, int* efficiency, int efficiencySize, int k){
    engineer_t *en = NULL;
    long long perf = 0;
    long long sum = 0; 

    if (n == 0) {
        return 0;
    }      
    queue_init();

    en = engineer_create(n, speed, efficiency);
    qsort(en, n, sizeof(engineer_t), com);

    for (int i = 0; i < n; i++) {
        long long effic_cur = en[i].efficiency;
        long long speed_cur = en[i].speed;
     
        perf = max(perf, (sum + speed_cur) * effic_cur);
        queue_push(speed_cur);
        sum += speed_cur;
        
        if (queue_size() >= k) {
            int speed_min;
            speed_min = queue_pop();
            sum -= speed_min;
        }
    }
    engineer_destory(en);

    return perf % ((int)1e9 + 7);
}
```
